/* ---------- State ---------- */
let state = loadState();

function loadState() {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return structuredClone(DEFAULTS);
    return mergeDeep(structuredClone(DEFAULTS), JSON.parse(raw));
  } catch (e) {
    return structuredClone(DEFAULTS);
  }
}

function saveState() {
  try { localStorage.setItem(STORAGE_KEY, JSON.stringify(state)); } catch (e) {}
}

function mergeDeep(a, b) {
  for (const k in b) {
    if (b[k] && typeof b[k] === "object" && !Array.isArray(b[k])) {
      a[k] = mergeDeep(a[k] || {}, b[k]);
    } else {
      a[k] = b[k];
    }
  }
  return a;
}

/* ---------- Helpers ---------- */
function fmtMoney(n) {
  if (!isFinite(n)) return "0 ₸";
  return n.toLocaleString("ru-RU").replace(/,/g, " ") + " ₸";
}
function fmtMln(n) {
  const m = n / 1_000_000;
  if (Math.abs(m) >= 10) return m.toFixed(0) + " млн ₸";
  return m.toFixed(2).replace(".", ",") + " млн ₸";
}
function fmtNum(n) {
  return Math.round(n).toLocaleString("ru-RU").replace(/,/g, " ");
}
// Правильное склонение "точка / точки / точек"
function pluralPoints(n) {
  n = Math.abs(n);
  const mod10 = n % 10;
  const mod100 = n % 100;
  if (mod100 >= 11 && mod100 <= 14) return "точек";
  if (mod10 === 1) return "точка";
  if (mod10 >= 2 && mod10 <= 4) return "точки";
  return "точек";
}
function fmtSigned(n) {
  return (n >= 0 ? "+" : "−") + fmtMoney(Math.abs(n));
}
function fmtSignedMln(n) {
  return (n >= 0 ? "+" : "−") + fmtMln(Math.abs(n));
}
function escapeHtml(s) {
  return String(s || "").replace(/[&<>"']/g, c => ({
    "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;"
  }[c]));
}
function ddmmyyyy(iso) {
  if (!iso) return "";
  const [y, m, d] = iso.split("-");
  return `${d}.${m}.${y}`;
}

/* ---------- Form ↔ state binding ---------- */
function bindForm() {
  // simple fields
  const fields = {
    "f-kpNumber":   ["meta", "kpNumber"],
    "f-date":       ["meta", "date"],
    "f-city":       ["meta", "city"],
    "f-partner":    ["partner", "name"],
    "f-lpr":        ["partner", "lpr"],
    "f-lprTitle":   ["partner", "lprTitle"],
    "f-points":     ["partner", "points"],
    "f-turnover":   ["business", "turnover"],
    "f-avgCheck":   ["business", "avgCheck"],
    "f-commission": ["business", "commission"],
    "f-setup":      ["setup"],
    "f-acquiring":  ["acquiring", "rate"],
    "f-manager":    ["manager", "name"],
    "f-initials":   ["manager", "initials"],
    "f-phone":      ["manager", "phone"]
  };

  for (const [id, path] of Object.entries(fields)) {
    const el = document.getElementById(id);
    if (!el) continue;

    // initial value
    setFromPath(el, path);

    el.addEventListener("input", () => {
      const v = el.type === "number" ? Number(el.value) : el.value;
      setPath(state, path, v);
      saveState();
      render();
    });
  }

  // products
  document.querySelectorAll(".product").forEach(row => {
    const key = row.dataset.key;
    if (key === "freedomAccount") return; // handled separately below
    const cb = row.querySelector("input[type=checkbox]");
    const priceInput = row.querySelector(`input[data-price="${key}"]`);
    const discInput  = row.querySelector(`input[data-discount="${key}"]`);
    cb.checked = state.products[key].enabled;
    priceInput.value = state.products[key].price;
    if (discInput) discInput.value = state.products[key].discount || "";
    row.classList.toggle("on", cb.checked);

    cb.addEventListener("change", () => {
      state.products[key].enabled = cb.checked;
      row.classList.toggle("on", cb.checked);
      saveState();
      render();
    });
    priceInput.addEventListener("input", () => {
      state.products[key].price = Number(priceInput.value || 0);
      saveState();
      render();
    });
    if (discInput) {
      discInput.addEventListener("input", () => {
        state.products[key].discount = Number(discInput.value || 0);
        saveState();
        render();
      });
    }
  });

  // freedom account toggle
  const freedomRow = document.querySelector('.product[data-key="freedomAccount"]');
  const freedomCb = document.getElementById("f-freedom");
  if (freedomCb && freedomRow) {
    freedomCb.checked = state.freedomAccount.enabled;
    freedomRow.classList.toggle("on", freedomCb.checked);
    freedomCb.addEventListener("change", () => {
      state.freedomAccount.enabled = freedomCb.checked;
      freedomRow.classList.toggle("on", freedomCb.checked);
      saveState();
      render();
    });
  }

  // toolbar
  document.getElementById("btn-print").addEventListener("click", () => window.print());
  document.getElementById("btn-pdf").addEventListener("click", downloadPdf);
  document.getElementById("btn-download").addEventListener("click", downloadHtml);
  document.getElementById("btn-reset").addEventListener("click", () => {
    if (!confirm("Сбросить все поля к значениям по умолчанию?")) return;
    localStorage.removeItem(STORAGE_KEY);
    state = structuredClone(DEFAULTS);
    initFormValues();
    render();
  });
}

function setPath(obj, path, value) {
  let o = obj;
  for (let i = 0; i < path.length - 1; i++) o = o[path[i]];
  o[path[path.length - 1]] = value;
}
function getPath(obj, path) {
  let o = obj;
  for (const k of path) o = o ? o[k] : undefined;
  return o;
}
function setFromPath(el, path) {
  const v = getPath(state, path);
  el.value = v ?? "";
}

    freedomCb.checked = state.freedomAccount.enabled;
    freedomRow.classList.toggle("on", freedomCb.checked);
  }
  const map = {
    "f-kpNumber":   state.meta.kpNumber,

    "f-city":       state.meta.city,

  for (const [id, v] of Object.entries(map)) {

    if (el) el.value = v ?? "";
  }
}

/* ---------- Calculations ---------- */
function isFreedomFree(key) {
  return state.freedomAccount?.enabled && FREEDOM_FREE_PRODUCTS.includes(key);
}
function effectivePrice(p, key) {
  if (isFreedomFree(key)) return 0;
  const base = p?.price ?? 0;
  const disc = p?.discount ?? 0;
  return Math.max(0, base - disc);
}

function compute() {
  const s = state;
  const points = Math.max(1, Number(s.partner?.points) || 1);
  const orders = s.business.avgCheck > 0 ? Math.round(s.business.turnover / s.business.avgCheck) : 0;
  const lossPerOrder = s.business.avgCheck * (s.business.commission / 100);
  const aggregatorLoss = orders * lossPerOrder;

  // Множитель: per-location продукты × points, сетевые (loyalty, brandedApp) × 1
  const productMult = (k) => isPerLocation(k) ? points : 1;

  const deliveryAbon = s.products.delivery.enabled
    ? effectivePrice(s.products.delivery, "delivery") * productMult("delivery")
    : 0;
  const monthlyAbon = Object.entries(s.products)
    .filter(([_, p]) => p.enabled)
    .reduce((sum, [k, p]) => sum + effectivePrice(p, k) * productMult(k), 0);

  const scenarioPcts = [0.30, 0.50, 0.70, 1.00];
  const scenarios = scenarioPcts.map(p => {
    const ordersOnOwn = Math.round(orders * p);
    const savings = ordersOnOwn * lossPerOrder;
    const net = savings - deliveryAbon;
    return {
      pct: Math.round(p * 100),
      orders: ordersOnOwn,
      savings,
      net,
      netYear: net * 12
    };
  });

  return {
    orders,
    lossPerOrder,
    aggregatorLoss,
    deliveryAbon,
    monthlyAbon,
    scenarios,
    full: scenarios[scenarios.length - 1],
    points,
    productMult
  };
}

function renderContact() {
  const name = state.manager.name?.trim() || "";

/* ---------- PDF export (html2canvas + jsPDF) ---------- */
const A4_W_PX = 794;   // A4 width  @ 96dpi
const A4_H_PX = 1123;  // A4 height @ 96dpi
const PDF_PAD = 44;    // vertical padding inside each page

function pdfStatus(text) {
  const el = document.getElementById("pdf-status");
  if (el) el.textContent = text;
}
function pdfOverlay(on) {
  document.getElementById("pdf-overlay").classList.toggle("on", on);
}

async function downloadPdf() {
  if (typeof html2canvas === "undefined" || !window.jspdf) {
    alert("Библиотеки для PDF не загрузились. Проверь интернет и обнови страницу (Ctrl+F5).");
    return;
  }

  const source = document.getElementById("preview");
  if (!source || !source.children.length) return;

  pdfOverlay(true);
  pdfStatus("Раскладываем по страницам…");
  await new Promise(r => setTimeout(r, 50));

  const stage = document.getElementById("pdf-stage");
  stage.innerHTML = "";

  // Measuring container — same width/padding as a real page
  const probe = document.createElement("div");
  probe.className = "preview-shell pdf-page";
  probe.style.height = "auto";
  stage.appendChild(probe);

  const maxContent = A4_H_PX - PDF_PAD * 2;
  const blocks = Array.from(source.children).map(n => n.cloneNode(true));

  // Pack blocks into pages
  const pages = [];
  let current = [];
  let currentH = 0;

  for (const block of blocks) {
    probe.innerHTML = "";
    probe.appendChild(block.cloneNode(true));
    const h = probe.scrollHeight - PDF_PAD * 2;

    if (h > maxContent) {
      // Block taller than a page — flush current, give it its own page
      if (current.length) { pages.push(current); current = []; currentH = 0; }
      pages.push([block]);
      continue;
    }
    if (currentH + h > maxContent && current.length) {
      pages.push(current);
      current = [];
      currentH = 0;
    }
    current.push(block);
    currentH += h;
  }
  if (current.length) pages.push(current);

  probe.remove();

  // Render each page
  const { jsPDF } = window.jspdf;
  const pdf = new jsPDF({ orientation: "portrait", unit: "px", format: [A4_W_PX, A4_H_PX] });

  try {
    for (let i = 0; i < pages.length; i++) {
      pdfStatus(`Рендерим страницу ${i + 1} из ${pages.length}…`);
      await new Promise(r => setTimeout(r, 30));

      const pageEl = document.createElement("div");
      pageEl.className = "preview-shell pdf-page";
      pageEl.style.minHeight = A4_H_PX + "px";
      pages[i].forEach(b => pageEl.appendChild(b));
      stage.appendChild(pageEl);

      const canvas = await html2canvas(pageEl, {
        scale: 2,
        backgroundColor: getComputedStyle(document.documentElement)
          .getPropertyValue("--f-bg").trim() || "#0a0f0d",
        logging: false,
        useCORS: true,
        windowWidth: A4_W_PX
      });

      const img = canvas.toDataURL("image/jpeg", 0.92);
      if (i > 0) pdf.addPage([A4_W_PX, A4_H_PX], "portrait");

      // Fit image to page width; clip height to page
      const imgH = (canvas.height / canvas.width) * A4_W_PX;
      pdf.addImage(img, "JPEG", 0, 0, A4_W_PX, Math.min(imgH, A4_H_PX));

      pageEl.remove();
    }

    pdfStatus("Сохраняем…");
    const safe = (state.partner.name || "КП").replace(/[\\/:*?"<>|]/g, "-");
    pdf.save(`КП — ${safe}.pdf`);
  } catch (err) {
    console.error(err);
    alert("Не удалось собрать PDF: " + err.message);
  } finally {
    stage.innerHTML = "";
    pdfOverlay(false);
  }
}

/* ---------- Download standalone HTML ---------- */
function downloadHtml() {
  const previewHtml = document.getElementById("preview").innerHTML;
  // Extract only the preview-related CSS by grabbing the inline <style> and
  // filtering to the .preview-shell scope. Simpler: inline a known stylesheet.
  const standalone = buildStandaloneHtml(previewHtml);
  const blob = new Blob([standalone], { type: "text/html;charset=utf-8" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  const safe = (state.partner.name || "kp").replace(/[\\/:*?"<>|]/g, "-");
  a.download = `КП — ${safe}.html`;
  document.body.appendChild(a);
  a.click();
  setTimeout(() => { URL.revokeObjectURL(url); a.remove(); }, 500);
}

let TOKENS = FALLBACK_TOKENS;
let VALUE_PROPS = FALLBACK_VALUE_PROPS;
let configsLoaded = false;

async function loadEnterpriseConfigs() {
  if (configsLoaded) return;
  const tryFetch = async (path, fallback) => {
    try {
      const r = await fetch(path, { cache: "no-store" });
      if (!r.ok) throw new Error(r.status);
      const j = await r.json();
      return j && typeof j === "object" ? j : fallback;
    } catch (e) {
      console.info(`[enterprise] ${path} недоступен, используются встроенные значения`);
      return fallback;
    }
  };
  const [t, v] = await Promise.all([
    tryFetch("./config/design-tokens.json", FALLBACK_TOKENS),
    tryFetch("./config/value-props.json", FALLBACK_VALUE_PROPS)
  ]);
  TOKENS = { ...FALLBACK_TOKENS, ...t,
    colors: { ...FALLBACK_TOKENS.colors, ...(t.colors || {}) },
    fonts:  { ...FALLBACK_TOKENS.fonts,  ...(t.fonts  || {}) },
    sizes:  { ...FALLBACK_TOKENS.sizes,  ...(t.sizes  || {}) },
    layout: { ...FALLBACK_TOKENS.layout, ...(t.layout || {}) } };
  VALUE_PROPS = { ...FALLBACK_VALUE_PROPS, ...v,
    products: { ...FALLBACK_VALUE_PROPS.products, ...(v.products || {}) } };
  configsLoaded = true;
}

/* ---------- Поиск логотипа партнёра ---------- */
function normalizeDomain(input) {
  if (!input) return "";
  let d = input.trim().toLowerCase();
  d = d.replace(/^https?:\/\//, "").replace(/^www\./, "").split("/")[0].split("?")[0];
  return /\.[a-z]{2,}$/.test(d) ? d : "";
}

async function urlToDataUrl(url) {
  const res = await fetch(url, { mode: "cors" });
  if (!res.ok) throw new Error("HTTP " + res.status);
  const blob = await res.blob();
  if (!blob.type.startsWith("image/")) throw new Error("не изображение");
  if (blob.size < 200) throw new Error("пустое изображение");
  return await new Promise((resolve, reject) => {
    const fr = new FileReader();
    fr.onload = () => resolve(fr.result);
    fr.onerror = reject;
    fr.readAsDataURL(blob);
  });
}

async function findPartnerLogo(name, domainRaw, onStatus) {
  const domain = normalizeDomain(domainRaw);
  const say = onStatus || (() => {});

  // 1. Clearbit — основной источник
  if (domain) {
    say("Ищем логотип в Clearbit…");
    try {
      return { dataUrl: await urlToDataUrl(`https://logo.clearbit.com/${domain}?size=512`), source: "Clearbit" };
    } catch (e) { /* дальше */ }

    // 2. Фавикон сайта
    say("Пробуем фавикон сайта…");
    try {
      const fav = `https://www.google.com/s2/favicons?sz=256&domain=${encodeURIComponent(domain)}`;
      return { dataUrl: await urlToDataUrl(fav), source: "Favicon" };
    } catch (e) { /* дальше */ }
  }

  // 3. Wikipedia / Wikimedia
  if (name) {
    say("Ищем в Wikipedia…");
    try {
      const api = `https://ru.wikipedia.org/w/api.php?action=query&format=json&origin=*&prop=pageimages&piprop=original&generator=search&gsrsearch=${encodeURIComponent(name)}&gsrlimit=1`;
      const r = await fetch(api);
      const j = await r.json();
      const pages = j?.query?.pages || {};
      const first = Object.values(pages)[0];
      const src = first?.original?.source;
      if (src) return { dataUrl: await urlToDataUrl(src), source: "Wikipedia" };
    } catch (e) { /* дальше */ }
  }

  return null;
}

  const ps = vp.pilotSlide;
  slides.push({

    badge: ps.label,
    title: ps.title,

              <div class="ps-contact-name">${escHtml(slide.manager.name || "—")}</div>
              <div class="ps-contact-role">Freedom Restaurants</div>
              <div class="ps-contact-phone">${escHtml(slide.manager.phone || "")}</div>
            </div>` : ""}
        </div>`;
      break;
  }

  const bgClass = (slide.type === "cover" || slide.type === "section" || slide.type === "closing")
    ? "ps-slide ps-slide-hero" : "ps-slide";

  return `<div class="${bgClass}">${chrome}<div class="ps-body">${body}</div>${foot}</div>`;
}

function showPptxPreview() {
  const modal = document.getElementById("pptx-modal");
  const stage = document.getElementById("pptx-slide-stage");
  if (!pptxDeck) return;
  const { partner, slides, logoDataUrl } = pptxDeck;
  pptxSlideIndex = Math.max(0, Math.min(pptxSlideIndex, slides.length - 1));
  stage.innerHTML = renderSlideHtml(slides[pptxSlideIndex], pptxSlideIndex, slides.length, partner, logoDataUrl);
  document.getElementById("pptx-counter").textContent = `${pptxSlideIndex + 1} / ${slides.length}`;
  document.getElementById("pptx-prev").disabled = pptxSlideIndex === 0;
  document.getElementById("pptx-next").disabled = pptxSlideIndex === slides.length - 1;
  modal.classList.add("on");
}

/* ---------- Генерация .pptx ---------- */
async function generatePptx() {
  if (!window.PptxGenJS) {
    alert("Библиотека pptxgenjs не загрузилась. Проверь интернет и обнови страницу (Ctrl+F5).");
    return;
  }
  if (!pptxDeck) return;

  const btn = document.getElementById("pptx-download");
  const oldText = btn.textContent;
  btn.disabled = true;
  btn.textContent = "Собираем…";

  try {
    const C = TOKENS.colors, F = TOKENS.fonts, S = TOKENS.sizes, L = TOKENS.layout;
    const { partner, slides, logoDataUrl } = pptxDeck;

    const pptx = new PptxGenJS();
    pptx.defineLayout({ name: "FR16x9", width: S.slideW, height: S.slideH });
    pptx.layout = "FR16x9";
    pptx.author = "Freedom Restaurants";
    pptx.company = "Freedom Restaurants";
    pptx.title = `КП — ${partner}`;

    const W = S.slideW, H = S.slideH, MX = L.marginX;

    /* --- переиспользуемые компоненты --- */
    const addBg = (sl, hero) => {
      sl.background = { color: hero ? C.bgSection : C.bgDark };
      // мягкое зелёное свечение слева сверху
      sl.addShape(pptx.ShapeType.ellipse, {
        x: -2.2, y: -2.6, w: 8.5, h: 6.5,
        fill: { color: C.bgGradientTo, transparency: 72 }, line: { type: "none" }
      });
    };

    const addBadge = (sl, text) => {
      if (!text) return;
      const w = Math.min(6.4, 0.42 + text.length * 0.088);
      sl.addShape(pptx.ShapeType.roundRect, {
        x: MX, y: L.badgeY, w, h: 0.36, rectRadius: 0.18,
        fill: { color: "FFFFFF", transparency: 94 },
        line: { color: C.green, width: 0.6, transparency: 55 }
      });
      sl.addShape(pptx.ShapeType.ellipse, {
        x: MX + 0.17, y: L.badgeY + 0.145, w: 0.075, h: 0.075,
        fill: { color: C.green }, line: { type: "none" }
      });
      sl.addText(text, {
        x: MX + 0.3, y: L.badgeY, w: w - 0.34, h: 0.36,
        fontFace: F.heading, fontSize: S.badge, bold: true, color: C.textWhite,
        charSpacing: 1.1, valign: "middle"
      });
    };

    const addFreedomMark = (sl) => {
      sl.addImage({ data: FREEDOM_LOGO, x: W - MX - 1.55, y: L.badgeY - 0.03, w: 0.36, h: 0.42 });
      sl.addText("FREEDOM", {
        x: W - MX - 1.13, y: L.badgeY, w: 1.13, h: 0.36,
        fontFace: F.heading, fontSize: 13, bold: true, color: C.textWhite, valign: "middle"
      });
    };

    const addFooter = (sl, idx, total) => {
      sl.addText(`${partner} · Freedom`, {
        x: MX, y: L.footerY, w: 5, h: 0.3,
        fontFace: F.body, fontSize: S.footer, color: C.textDim
      });
      sl.addText(`${String(idx + 1).padStart(2, "0")} / ${String(total).padStart(2, "0")}`, {
        x: W - MX - 1.2, y: L.footerY, w: 1.2, h: 0.3,
        fontFace: F.body, fontSize: S.footer, color: C.textDim, align: "right"
      });
    };

    const addUnderline = (sl, y, w) => {
      sl.addShape(pptx.ShapeType.rect, {
        x: MX, y, w: w || L.underlineW, h: L.underlineH,
        fill: { color: C.orange }, line: { type: "none" }
      });
    };

    const addTitle = (sl, text, y) => {
      sl.addText(text, {
        x: MX, y: y ?? L.titleY, w: W - MX * 2, h: 0.95,
        fontFace: F.heading, fontSize: S.titleSlide, bold: true,
        color: C.textWhite, valign: "top", lineSpacingMultiple: 1.05
      });
    };

    const numberedCard = (sl, o) => {
      sl.addShape(pptx.ShapeType.roundRect, {
        x: o.x, y: o.y, w: o.w, h: o.ch, rectRadius: L.cardRadius,
        fill: { color: C.cardDark }, line: { color: C.green, width: 0.9, transparency: 45 }
      });
      sl.addText(o.num, {
        x: o.x + 0.18, y: o.y + 0.14, w: o.w - 0.36, h: 0.5,
        fontFace: F.heading, fontSize: 24, bold: true, color: C.green
      });
      sl.addText(o.head, {
        x: o.x + 0.18, y: o.y + 0.66, w: o.w - 0.36, h: 0.5,
        fontFace: F.heading, fontSize: S.cardTitle - 2, bold: true, color: C.textWhite, valign: "top"
      });
      sl.addText(o.t, {
        x: o.x + 0.18, y: o.y + 1.14, w: o.w - 0.36, h: o.ch - 1.3,
        fontFace: F.body, fontSize: S.cardText, color: C.textMuted, valign: "top"
      });
    };

    const iconCard = (sl, o) => {
      sl.addShape(pptx.ShapeType.roundRect, {
        x: o.x, y: o.y, w: o.w, h: o.ch, rectRadius: L.cardRadius,
        fill: { color: C.cardDark }, line: { color: C.green, width: 0.9, transparency: 45 }
      });
      sl.addShape(pptx.ShapeType.ellipse, {
        x: o.x + 0.2, y: o.y + 0.22, w: 0.17, h: 0.17,
        fill: { color: C.green }, line: { type: "none" }
      });
      sl.addText(o.head, {
        x: o.x + 0.2, y: o.y + 0.5, w: o.w - 0.4, h: 0.55,
        fontFace: F.heading, fontSize: S.cardTitle - 2, bold: true, color: C.textWhite, valign: "top"
      });
      sl.addText(o.t, {
        x: o.x + 0.2, y: o.y + 1.05, w: o.w - 0.4, h: o.ch - 1.2,
        fontFace: F.body, fontSize: S.cardText, color: C.textMuted, valign: "top"
      });
    };

    const stepFlow = (sl, steps, y) => {
      if (!steps || !steps.length) return;
      const total = W - MX * 2;
      const cell = total / steps.length;
      steps.forEach((s, i) => {
        const cx = MX + cell * i;
        sl.addShape(pptx.ShapeType.ellipse, {
          x: cx, y, w: 0.42, h: 0.42,
          fill: { color: C.bgDark }, line: { color: C.green, width: 1.4 }
        });
        sl.addText(String(i + 1), {
          x: cx, y, w: 0.42, h: 0.42,
          fontFace: F.heading, fontSize: 12, bold: true, color: C.green,
          align: "center", valign: "middle"
        });
        sl.addText(s, {
          x: cx + 0.5, y: y + 0.03, w: cell - 0.75, h: 0.36,
          fontFace: F.body, fontSize: 11, color: C.textWhite, valign: "middle"
        });
        if (i < steps.length - 1) {
          sl.addShape(pptx.ShapeType.rect, {
            x: cx + cell - 0.22, y: y + 0.2, w: 0.16, h: 0.02,
            fill: { color: C.green, transparency: 45 }, line: { type: "none" }
          });
        }
      });
    };

    const chevronFlow = (sl, steps, y) => {
      if (!steps || !steps.length) return;
      const total = W - MX * 2;
      const cell = total / steps.length;
      steps.forEach((s, i) => {
        const t = 62 - i * 15;
        sl.addShape(pptx.ShapeType.chevron, {
          x: MX + cell * i, y, w: cell - 0.06, h: 0.52,
          fill: { color: C.green, transparency: Math.max(0, t) },
          line: { type: "none" }
        });
        sl.addText(s, {
          x: MX + cell * i + 0.16, y, w: cell - 0.36, h: 0.52,
          fontFace: F.heading, fontSize: 11, bold: true,
          color: i >= 2 ? C.onGreen : C.textWhite, align: "center", valign: "middle"
        });
      });
    };

    /* --- сборка слайдов --- */
    slides.forEach((sd, idx) => {
      const hero = sd.type === "cover" || sd.type === "section" || sd.type === "closing";
      const sl = pptx.addSlide();
      addBg(sl, hero);
      addBadge(sl, sd.badge);
      addFreedomMark(sl);
      addFooter(sl, idx, slides.length);

      if (sd.type === "cover") {
        if (logoDataUrl) {
          sl.addImage({ data: logoDataUrl, x: MX, y: 1.15, w: 1.5, h: 0.95, sizing: { type: "contain", w: 1.5, h: 0.95 } });
        }
        sl.addText(sd.partner, {
          x: MX, y: logoDataUrl ? 2.3 : 2.05, w: W - MX * 2, h: 1.2,
          fontFace: F.heading, fontSize: S.titleCover, bold: true, color: C.textWhite
        });
        sl.addShape(pptx.ShapeType.ellipse, {
          x: MX, y: logoDataUrl ? 3.62 : 3.37, w: 0.17, h: 0.17,
          fill: { color: C.green }, line: { type: "none" }
        });
        sl.addText(sd.subtitle, {
          x: MX + 0.3, y: logoDataUrl ? 3.4 : 3.15, w: W - MX * 2, h: 0.6,
          fontFace: F.heading, fontSize: 30, bold: true, color: C.green
        });
        addUnderline(sl, logoDataUrl ? 4.18 : 3.93, 5.2);
        sl.addText(sd.tagline, {
          x: MX, y: logoDataUrl ? 4.42 : 4.17, w: 7.6, h: 0.9,
          fontFace: F.body, fontSize: 16, color: C.textMuted, lineSpacingMultiple: 1.25
        });

      } else if (sd.type === "section") {
        sl.addText(sd.title.replace(/\n/g, "\n"), {
          x: MX, y: 2.1, w: W - MX * 2, h: 2.1,
          fontFace: F.heading, fontSize: S.titleSection, bold: true,
          color: C.textWhite, lineSpacingMultiple: 1.05
        });
        addUnderline(sl, 4.35, 4.6);

      } else if (sd.type === "numbered") {
        addTitle(sl, sd.title);
        const n = sd.cards.length;
        const gap = L.cardGap;
        const cw = (W - MX * 2 - gap * (n - 1)) / n;
        sd.cards.forEach((c, i) => {
          numberedCard(sl, {
            x: MX + (cw + gap) * i, y: L.contentY, w: cw, ch: 2.5,
            num: String(i + 1).padStart(2, "0"), head: c.h, t: c.t
          });
        });
        if (sd.outro) {
          sl.addText(sd.outro, {
            x: MX, y: 5.45, w: W - MX * 2, h: 0.8,
            fontFace: F.heading, fontSize: 15, bold: true, color: C.textWhite, lineSpacingMultiple: 1.2
          });
        }
        if (sd.footnote) {
          sl.addText("* " + sd.footnote, {
            x: MX, y: 6.45, w: W - MX * 2, h: 0.4,
            fontFace: F.body, fontSize: 9, italic: true, color: C.textDim
          });
        }

      } else if (sd.type === "value" || sd.type === "iconCards") {
        addTitle(sl, sd.title);
        if (sd.lead) {
          sl.addText(sd.lead, {
            x: MX, y: L.titleY + 0.92, w: W - MX * 2 - 1, h: 0.6,
            fontFace: F.body, fontSize: S.lead, color: C.textMuted, lineSpacingMultiple: 1.2
          });
        }
        const n = Math.min(sd.cards.length, 4);
        const gap = L.cardGap;
        const cw = (W - MX * 2 - gap * (n - 1)) / n;
        const cy = sd.lead ? L.contentY + 0.25 : L.contentY;
        sd.cards.slice(0, 4).forEach((c, i) => {
          iconCard(sl, { x: MX + (cw + gap) * i, y: cy, w: cw, ch: 2.35, head: c.h, t: c.t });
        });
        if (sd.flow?.length) stepFlow(sl, sd.flow, cy + 2.62);

      } else if (sd.type === "nowAfter") {
        addTitle(sl, sd.title);
        const halfW = (W - MX * 2 - 0.3) / 2;
        // Сейчас
        sl.addShape(pptx.ShapeType.roundRect, {
          x: MX, y: L.contentY, w: halfW, h: 2.35, rectRadius: L.cardRadius,
          fill: { color: C.cardDark2, transparency: 25 }, line: { color: C.textDim, width: 0.75, transparency: 55 }
        });
        sl.addText("СЕЙЧАС", { x: MX + 0.25, y: L.contentY + 0.2, w: halfW - 0.5, h: 0.3,
          fontFace: F.heading, fontSize: 10, bold: true, color: C.textMuted, charSpacing: 1.2 });
        sl.addText(sd.now.value, { x: MX + 0.25, y: L.contentY + 0.55, w: halfW - 0.5, h: 0.75,
          fontFace: F.heading, fontSize: S.bigNumber, bold: true, color: C.textWhite });
        sl.addText(sd.now.caption, { x: MX + 0.25, y: L.contentY + 1.3, w: halfW - 0.5, h: 0.4,
          fontFace: F.body, fontSize: 12, color: C.textMuted });
        sl.addText(sd.now.note, { x: MX + 0.25, y: L.contentY + 1.72, w: halfW - 0.5, h: 0.55,
          fontFace: F.body, fontSize: 9.5, color: C.textDim, lineSpacingMultiple: 1.15 });
        // После
        const ax = MX + halfW + 0.3;
        sl.addShape(pptx.ShapeType.roundRect, {
          x: ax, y: L.contentY, w: halfW, h: 2.35, rectRadius: L.cardRadius,
          fill: { color: C.cardDark }, line: { color: C.green, width: 1.3 }
        });
        sl.addText("ПОСЛЕ ПЕРЕХОДА В FREEDOM SUPER APP", { x: ax + 0.25, y: L.contentY + 0.2, w: halfW - 0.5, h: 0.3,
          fontFace: F.heading, fontSize: 10, bold: true, color: C.green, charSpacing: 1.1 });
        sl.addText(sd.after.value, { x: ax + 0.25, y: L.contentY + 0.55, w: halfW - 0.5, h: 0.75,
          fontFace: F.heading, fontSize: S.bigNumber, bold: true, color: C.green });
        sl.addText(sd.after.caption, { x: ax + 0.25, y: L.contentY + 1.3, w: halfW - 0.5, h: 0.4,
          fontFace: F.body, fontSize: 12, color: C.textMuted });
        sl.addText(sd.after.note, { x: ax + 0.25, y: L.contentY + 1.72, w: halfW - 0.5, h: 0.55,
          fontFace: F.body, fontSize: 9.5, color: C.textDim, lineSpacingMultiple: 1.15 });

        addUnderline(sl, 5.42, W - MX * 2);
        chevronFlow(sl, sd.flow, 5.62);
        if (sd.footnote) {
          sl.addText("* " + sd.footnote, { x: MX, y: 6.42, w: W - MX * 2, h: 0.35,
            fontFace: F.body, fontSize: 9, italic: true, color: C.textDim });

      } else if (sd.type === "terms") {
        addTitle(sl, sd.title);
        const hw = (W - MX * 2 - 0.3) / 2;
        sd.highlights.forEach((hl, i) => {
          const hx = MX + (hw + 0.3) * i;
          sl.addShape(pptx.ShapeType.roundRect, {
            x: hx, y: L.contentY - 0.25, w: hw, h: 1.32, rectRadius: L.cardRadius,
            fill: { color: C.cardDark }, line: { color: C.green, width: 1.1 }
          });
          sl.addText(hl.value, { x: hx + 0.25, y: L.contentY - 0.12, w: hw - 0.5, h: 0.62,

          sl.addText(hl.label, { x: hx + 0.25, y: L.contentY + 0.52, w: hw - 0.5, h: 0.28,
            fontFace: F.heading, fontSize: 12, bold: true, color: C.textWhite });
          sl.addText(hl.note, { x: hx + 0.25, y: L.contentY + 0.78, w: hw - 0.5, h: 0.4,
            fontFace: F.body, fontSize: 9.5, color: C.textMuted, lineSpacingMultiple: 1.1 });
        });

        const rows = [[
          { text: "ПОЗИЦИЯ", options: { bold: true, color: C.textMuted, fontSize: 9.5, fill: { color: C.bgSection } } },

          { text: r.label, options: { color: C.textWhite, fontSize: 11 } },

          fill: { color: C.bgDark }, rowH: 0.33, valign: "middle", margin: 0.08

      } else if (sd.type === "showcase") {

        sl.addText(sd.lead, {
          x: MX, y: L.titleY + 0.9, w: W - MX * 2, h: 0.4,

        const colW = 3.55;
        const phoneW = 2.0, phoneH = 3.6;
        const phoneX = (W - phoneW) / 2;
        sd.left.forEach((x, i) => {
          sl.addText(x.h, { x: MX, y: L.contentY + i * 0.82, w: colW, h: 0.28,
            fontFace: F.heading, fontSize: 12, bold: true, color: C.textWhite });
          sl.addText(x.t, { x: MX, y: L.contentY + i * 0.82 + 0.27, w: colW, h: 0.45,
            fontFace: F.body, fontSize: 10, color: C.textMuted, lineSpacingMultiple: 1.1 });

    btn.textContent = oldText;
  }
}

/* ---------- Обработчики Enterprise-режима ---------- */
function setPartnerMode(mode) {
  state.mode = mode;
  saveState();
  document.querySelectorAll(".mode-card").forEach(el => {
    el.classList.toggle("on", el.dataset.mode === mode);
  });
  document.getElementById("enterprise-fields").style.display = mode === "enterprise" ? "" : "none";
  document.getElementById("btn-pptx").style.display = mode === "enterprise" ? "" : "none";
  document.getElementById("btn-pdf").style.display = mode === "enterprise" ? "none" : "";
  document.getElementById("btn-print").style.display = mode === "enterprise" ? "none" : "";
  const note = document.getElementById("enterprise-note");
  if (note) note.style.display = mode === "enterprise" ? "" : "none";
}

async function handleLogoSearch() {
  const statusEl = document.getElementById("logo-status");
  const btn = document.getElementById("btn-logo-search");
  btn.disabled = true;
  statusEl.textContent = "Ищем логотип…";
  statusEl.className = "logo-status";
  try {
    const found = await findPartnerLogo(
      state.partner.name,
      state.enterprise?.domain,
      msg => { statusEl.textContent = msg; }
    );
    if (found) {
      state.enterprise = state.enterprise || {};
      state.enterprise.logoDataUrl = found.dataUrl;
      state.enterprise.logoSource = found.source;
      saveState();
      updateLogoPreview();
      statusEl.textContent = `Найден через ${found.source}. Если не подходит — загрузи свой файл.`;
      statusEl.className = "logo-status ok";
    } else {
      statusEl.textContent = "Автоматически не нашли — загрузи файл вручную ниже.";
      statusEl.className = "logo-status warn";
    }
  } catch (e) {
    statusEl.textContent = "Поиск не удался: " + e.message + ". Загрузи файл вручную.";
    statusEl.className = "logo-status warn";
  } finally {
    btn.disabled = false;
  }
}

function updateLogoPreview() {
  const box = document.getElementById("logo-preview");
  const url = state.enterprise?.logoDataUrl;
  if (url) {
    box.innerHTML = `<img src="${url}" alt="Логотип партнёра" />`;
    box.classList.add("has");
  } else {
    box.innerHTML = `<span>Логотип не выбран</span>`;
    box.classList.remove("has");
  }
}

function bindEnterprise() {
  document.querySelectorAll(".mode-card").forEach(el => {
    el.addEventListener("click", () => setPartnerMode(el.dataset.mode));
  });

  const domainInput = document.getElementById("f-domain");
  if (domainInput) {
    domainInput.addEventListener("input", e => {
      state.enterprise = state.enterprise || {};
      state.enterprise.domain = e.target.value;
      saveState();
    });
  }

  document.getElementById("btn-logo-search").addEventListener("click", handleLogoSearch);

  document.getElementById("f-logo-file").addEventListener("change", e => {
    const file = e.target.files?.[0];
    if (!file) return;
    const fr = new FileReader();
    fr.onload = () => {
      state.enterprise = state.enterprise || {};
      state.enterprise.logoDataUrl = fr.result;
      state.enterprise.logoSource = "Загружен вручную";
      saveState();
      updateLogoPreview();
      const st = document.getElementById("logo-status");
      st.textContent = "Логотип загружен.";
      st.className = "logo-status ok";
    };
    fr.readAsDataURL(file);
  });

  document.getElementById("btn-logo-clear").addEventListener("click", () => {
    if (state.enterprise) {
      state.enterprise.logoDataUrl = null;
      state.enterprise.logoSource = null;
    }
    saveState();
    updateLogoPreview();
    document.getElementById("logo-status").textContent = "";
  });

  document.getElementById("btn-pptx").addEventListener("click", async () => {
    await loadEnterpriseConfigs();
    pptxDeck = buildPresentationStructure();
    pptxSlideIndex = 0;
    showPptxPreview();
  });

  document.getElementById("pptx-prev").addEventListener("click", () => { pptxSlideIndex--; showPptxPreview(); });
  document.getElementById("pptx-next").addEventListener("click", () => { pptxSlideIndex++; showPptxPreview(); });
  document.getElementById("pptx-close").addEventListener("click", () => {
    document.getElementById("pptx-modal").classList.remove("on");
  });
  document.getElementById("pptx-download").addEventListener("click", generatePptx);

  document.addEventListener("keydown", e => {
    const modal = document.getElementById("pptx-modal");
    if (!modal.classList.contains("on")) return;
    if (e.key === "ArrowLeft" && pptxSlideIndex > 0) { pptxSlideIndex--; showPptxPreview(); }
    if (e.key === "ArrowRight" && pptxDeck && pptxSlideIndex < pptxDeck.slides.length - 1) { pptxSlideIndex++; showPptxPreview(); }
    if (e.key === "Escape") modal.classList.remove("on");
  });

  setPartnerMode(state.mode || "merchant");
  updateLogoPreview();
  loadEnterpriseConfigs();
}

function init() {
  const appLogo = document.getElementById("app-logo");
  if (appLogo) appLogo.innerHTML = `<img src="${FREEDOM_LOGO}" alt="Freedom" />`;
  initFormValues();
  bindForm();
  bindEnterprise();
  render();
}

document.addEventListener("DOMContentLoaded", init);
